Public Function 配列比較差分取得(a As Variant, b As Variant) As Variant
 
Dim i As Long
Dim j As Long
Dim Q As Long
Dim r As Long, n As Long
'Dim a() As Variant
'Dim b() As Variant
Dim c() As Variant
Dim msg1 As String
Dim msg2 As String
Dim result As Variant
Dim buf As Variant
    '対象文字
'    a = Array("みかん", "めろん", "いちご", "きゃべつ", "れもん")
'    b = Array("りんご", "じゃがいも", "さくらんぼ", "めろん", "いちご", "れもん", "ごぼう")
For r = 1 To 2
  Q = 1
  ReDim Preserve c(Q)
  If r = 1 Then
      buf = a
  Else
      buf = b
  End If
  '配列をループ
  For i = LBound(buf) To UBound(buf)

      '配列を比較
      If r = 1 Then
          result = Filter(b, a(i))
      Else
          result = Filter(a, b(i))
      End If

      '片側にあって、片側にないものを取得
      If UBound(result) = -1 Then
          ReDim Preserve c(Q)
          c(Q) = buf(i)
          Q = Q + 1
      End If
  Next i
  ReDim temp(1 To UBound(c))
  For j = 1 To UBound(c)
      If r = 1 Then
          msg1 = msg1 & c(j) & vbCrLf
      Else
          n = n + 1
          temp(n) = c(j)
          msg2 = msg2 & c(j) & "_"
      End If
  Next j
  Erase c
Next r
'MsgBox "【AにあってBに無い】" & vbCrLf & msg1 & _
'    vbCrLf & "【BにあってAに無い】" & vbCrLf & msg2
配列比較差分取得 = temp
End Function

Public Function duplicateDelete(s As String) As String '同じセル内で同じ文字を削除

D = Split(s, " ")

Public Function DB_Materials(CoilText As Double, coilNo As Long) As Single 'コイル本数、
  Dim i As Long

  Dim FPath As String: FPath = PATH_AIM_参照
  Dim Target As String: Target = "梱包資材マスタ.accdb"
  
  Dim filePath As String: filePath = FPath & Target
  Dim SQLfilter As String: SQLfilter = ""
  Dim TempHiki As Variant: TempHiki = GetRecordsArr(filePath, tableName, SQLfilter)

    MsgBox "資材重量なし", vbCritical
    Exit Function

  '====== 資材重量の計算 ======

  materials = CalculateMaterials(CoilText, coilNo, TempHiki, fieldIdx)

  '====== 表記用フォームへの反映 ======
  SetDisplayFormValues materials, coilNo, CoilText
  
  '====== ListViewにデータを表示 ======
  PopulateListViewMaterials TempHiki, fieldIdx
  
  '====== シートへの転記 ======
  WriteToSheet coilNo, CoilText, materials
  
  '====== シート装飾 ======
  FormatSheet
  
End Function

Public Function GetFieldIndices(Hikifields As Variant) As MaterialFieldIndices
  Dim idx As MaterialFieldIndices
  Dim i As Long
  
  For i = LBound(Hikifields, 2) To UBound(Hikifields, 2)
    Select Case Hikifields(1, i)
      Case "管理番号": idx.管理番号 = i
      Case "梱包資材名": idx.梱包資材名 = i
      Case "単位質量": idx.単位質量 = i
      Case "係数": idx.係数 = i
    End Select
  Next i
  
  GetFieldIndices = idx
End Function

'========== 3. 資材重量計算 ==========
Private Function CalculateMaterials(CoilText As Double, coilNo As Long, _
                                     TempHiki As Variant, fieldIdx As MaterialFieldIndices) As MaterialWeights
  Dim mat As MaterialWeights
  Dim T1 As Double, T2 As Double
  
  '========== コイル巾と高さの計算 ==========
  SetCoilDimensions mat, CoilText, coilNo
  
  '========== チップボールサイズの決定 ==========
  SetDiameterInfo mat
  
  '========== 各資材重量の計算 ==========
  mat.PIN = CalculatePIN(TempHiki, fieldIdx, mat)
  mat.TIP = CalculateTIP(CoilText, mat, TempHiki, fieldIdx)
  mat.SUT = CalculateSUT(mat, TempHiki, fieldIdx)
  mat.ESA = CalculateESA(mat, TempHiki, fieldIdx)
  mat.POR = CalculatePOR(mat, TempHiki, fieldIdx)
  mat.HB = DB_HB(CoilText, coilNo)
  mat.PET = CalculatePET(mat, TempHiki, fieldIdx)
  mat.PAR = CalculatePAR(TempHiki, fieldIdx, mat)
  mat.DRY = CalculateDRY(TempHiki, fieldIdx, mat)
  
  '========== NW（製品重量）の計算 ==========
  GetCoilWeights T1, T2, coilNo '引数で渡された値を直接書き換える ByRef（参照渡し）
  mat.NW = CalculateNW(CoilText, coilNo, T1, T2)
  
  '========== 風袋・GWの計算 ==========
  mat.HU = mat.PIN + mat.TIP + mat.SUT + mat.ESA + mat.POR + mat.HB + mat.PET + mat.PAR + mat.DRY
  mat.GW = mat.HU + mat.NW
  
  CalculateMaterials = mat
End Function

'========== 4. コイル寸法の設定 ==========
Public Sub SetCoilDimensions(ByRef mat As MaterialWeights, CoilText As Double, coilNo As Long)
  Dim KTA As Single

  mat.col = GetCoilWidth()

  With テスラ指定サイズ
    '本数が空白の場合終了
    If .CoilH1 = "" And .CoilH2 = "" And .CoilH3 = "" And .CoilH4 = "" Then
      .NW1 = "": .GW1 = "": .TA1 = ""
      .NW2 = "": .GW2 = "": .TA2 = ""
      .NW3 = "": .GW3 = "": .TA3 = ""
      .NW4 = "": .GW4 = "": .TA4 = ""
      Exit Sub
    End If

    '高さ計算（共通関数へ集約）
    mat.TA = GetCoilTA(CoilText, mat.col)
'    KTA = val((mat.TA * 1000) + 150 + 150) + (2.35 * 2)
    KTA = val((mat.TA * 1000) + PALLET_HEIGHT_MM * 2) + (HB_THICKNESS_MM * 2)
    
    Select Case coilNo
      Case 1: .TA1 = Format(Application.WorksheetFunction.RoundUp(KTA, -1), "0.0")
      Case 2: .TA2 = Format(Application.WorksheetFunction.RoundUp(KTA, -1), "0.0")
      Case 3: .TA3 = Format(Application.WorksheetFunction.RoundUp(KTA, -1), "0.0")
      Case 4: .TA4 = Format(Application.WorksheetFunction.RoundUp(KTA, -1), "0.0")
    End Select
  End With
End Sub

'========== 5. 直径情報の設定 ==========円周の公式:円周 = 直径 × 円周率
Public Sub SetDiameterInfo(ByRef mat As MaterialWeights)
  With テスラ指定サイズ
    If .TIP1000 Then
      mat.TempDiameter = "1100": mat.HalfDiameter = "550"
    ElseIf .TIP950 Then
      mat.TempDiameter = "1000": mat.HalfDiameter = "500"
    ElseIf .TIP820 Then
      mat.TempDiameter = "900":  mat.HalfDiameter = "450"
    End If
  End With
  mat.GAI = GetCoilGAI()
End Sub

'========== 各資材の重量計算関数==========
Public Function CalculatePIN(TempHiki As Variant, fieldIdx As MaterialFieldIndices, ByRef mat As MaterialWeights) As Double
  Dim 単重 As Variant: 単重 = GetUnitMassAndCoefficient(TempHiki, "テスラピン", fieldIdx.梱包資材名, fieldIdx.単位質量, fieldIdx.係数)
  mat.pin式 = 単重
  Dim result As Double: result = val(4 * 単重(0) * 単重(1))
  表記用.PIsiki = "【ピン】本数×単重" & vbCrLf & "4本 × " & Format(単重(0), "0.000") & " = " & Format(result, "0.000")
  CalculatePIN = result
End Function

'ﾁｯﾌﾟﾎﾞｰﾙの高さ　1枚0.7mm’ﾁｯﾌﾟﾎﾞｰﾙ数は積み本数+1_重量
Public Function CalculateTIP(CoilText As Double, mat As MaterialWeights, TempHiki As Variant, fieldIdx As MaterialFieldIndices) As Double
  Dim 単重 As Variant
  With テスラ指定サイズ
    If .TIP1000 Then
      単重 = GetUnitMassAndCoefficient(TempHiki, "チップボール1000ф", fieldIdx.梱包資材名, fieldIdx.単位質量, fieldIdx.係数)
      mat.チップボール式 = 単重
    ElseIf .TIP950 Then
      単重 = GetUnitMassAndCoefficient(TempHiki, "チップボール950ф", fieldIdx.梱包資材名, fieldIdx.単位質量, fieldIdx.係数)
      mat.チップボール式 = 単重
    ElseIf .TIP820 Then
      単重 = GetUnitMassAndCoefficient(TempHiki, "チップボール820ф", fieldIdx.梱包資材名, fieldIdx.単位質量, fieldIdx.係数)
      mat.チップボール式 = 単重
    End If
  End With

  Dim maisu As Double: maisu = CoilText + CHIP_EXTRA_SHEETS
  Dim result As Double: result = val(maisu * 単重(0) * 単重(1))
  表記用.TIPsiki = "【チップボール】（積み本数＋" & CHIP_EXTRA_SHEETS & "枚）×単重" & vbCrLf & _
    "(" & Format(CoilText, "0") & " + " & CHIP_EXTRA_SHEETS & ") × " & Format(単重(0), "0.000") & " = " & Format(result, "0.00")
  CalculateTIP = result
End Function

'5周は巻く
Public Function CalculateSUT(mat As MaterialWeights, TempHiki As Variant, fieldIdx As MaterialFieldIndices) As Double
  Dim 単重 As Variant: 単重 = GetUnitMassAndCoefficient(TempHiki, "ストレッチフィルム", fieldIdx.梱包資材名, fieldIdx.単位質量, fieldIdx.係数)
  mat.SUT式 = 単重
  Dim result As Double: result = val(mat.GAI * STRETCH_WRAPS * 単重(0) * 単重(1))
  表記用.SUTsiki = "【ストレッチフィルム】外周" & STRETCH_WRAPS & "周分の長さ×単重" & vbCrLf & _
    Format(mat.GAI, "0.000") & "m × " & STRETCH_WRAPS & "周 × " & Format(単重(0), "0.000") & " = " & Format(result, "0.000")
  CalculateSUT = result
End Function

'エサフォームの高さが550　1.5倍ほどの高さになった時2周とする
Public Function CalculateESA(mat As MaterialWeights, TempHiki As Variant, fieldIdx As MaterialFieldIndices) As Double
  Dim 単重 As Variant: 単重 = GetUnitMassAndCoefficient(TempHiki, "エサフォーム", fieldIdx.梱包資材名, fieldIdx.単位質量, fieldIdx.係数)
  mat.ESA式 = 単重
  Dim result As Double
  Dim formula As String
  If mat.TA > ESA_DOUBLE_LIMIT_M Then
    result = val(mat.GAI * 2 * 単重(0) * 単重(1))
    formula = "【エサフォーム】外周の長さ×単重（2周）" & vbCrLf & Format(mat.GAI, "0.000") & "m × 2周 × " & Format(単重(0), "0.000") & " = " & Format(result, "0.000")
  Else
    result = val(mat.GAI * 単重(0) * 単重(1))
    formula = "【エサフォーム】外周の長さ×単重（1周）" & vbCrLf & Format(mat.GAI, "0.000") & "m × 1周 × " & Format(単重(0), "0.000") & " = " & Format(result, "0.000")
  End If
  表記用.ESAsiki = formula
  CalculateESA = result
End Function

 '高さﾌﾟﾗｽ分で包む　ﾌﾟﾗｽ分は高さの半分とする '同部分
Public Function CalculatePOR(mat As MaterialWeights, TempHiki As Variant, fieldIdx As MaterialFieldIndices) As Double
  Dim 単重 As Variant: 単重 = GetUnitMassAndCoefficient(TempHiki, "ポリシート", fieldIdx.梱包資材名, fieldIdx.単位質量, fieldIdx.係数)
  mat.POR式 = 単重
  Dim AREA As Double, AREA2 As Double, PIWork As Double
  PIWork = WorksheetFunction.PI()
  Dim 半径 As Long
  With テスラ指定サイズ
    If .TIP1000 Then
      AREA = val(550 ^ 2 * PIWork)
      半径 = 550
    ElseIf .TIP950 Then
      AREA = val(500 ^ 2 * PIWork)
      半径 = 500
    ElseIf .TIP820 Then
      AREA = val(450 ^ 2 * PIWork)
      半径 = 450
    End If
  End With
  AREA2 = val((mat.TA + mat.TA / 2) * mat.GAI) 'ややプラスしている
  '「底面の円の面積2つ」と「側面の四角形の面積」を足す
  Dim result As Double: result = val((AREA * 2 + AREA2) / 10 ^ 6 * 単重(0) * 単重(1))
  表記用.PORsiki = "【ポリシート】円柱の表面積＝2πr^2＋2πrh" & vbCrLf & "(底面:2π×" & 半径 & "2 + 側面:2π×" & 半径 & "×" & Format(mat.TA, "0.000") & ") × " & Format(単重(0), "0.000") & " = " & Format(result, "0.000")
  CalculatePOR = result
End Function

'ペットバンド胴巻き4本縦バンド4本締める時の余分を200mm
Public Function CalculatePET(mat As MaterialWeights, TempHiki As Variant, fieldIdx As MaterialFieldIndices) As Double
  Dim 単重 As Variant: 単重 = GetUnitMassAndCoefficient(TempHiki, "PETバンド", fieldIdx.梱包資材名, fieldIdx.単位質量, fieldIdx.係数)
  mat.PET式 = 単重
  Dim DOU As Double, TATE As Double
  DOU = val((mat.GAI + BAND_SLACK_M) * 4)
  With テスラ指定サイズ
    If .TIP1000 Then
      TATE = val((((mat.TA + BAND_SLACK_M) * 4) * 2) + ((1100 * 4) * 2) / 10 ^ 3)
    ElseIf .TIP950 Then
      TATE = val((((mat.TA + BAND_SLACK_M) * 4) * 2) + ((1000 * 4) * 2) / 10 ^ 3)
    ElseIf .TIP820 Then
      TATE = val((((mat.TA + BAND_SLACK_M) * 4) * 2) + ((900 * 4) * 2) / 10 ^ 3)
    End If
  End With
  Dim result As Double: result = (DOU + TATE) * 単重(0) * 単重(1)
  表記用.PETsiki = "【PETバンド】胴巻き4本分＋縦バンド4本分" & vbCrLf & "胴巻き:" & Format(DOU, "0.000") & "m + 縦バンド:" & Format(TATE, "0.000") & "m = " & Format(result, "0.000")
  CalculatePET = result
End Function

'========== 主関数（簡潔版）==========
Public Function DB_HB(CoilText As Double, coilNo As Long) As Double
  If Not IsNumeric(coilNo) Then Exit Function
  
  Dim i As Long
  Dim tableName As String: tableName = "資材重量"
  Dim FPath As String: FPath = PATH_AIM_参照
  Dim Target As String: Target = "梱包資材マスタ.accdb"
  
  Dim filePath As String: filePath = FPath & Target
  Dim SQLfilter As String: SQLfilter = ""
  Dim TempHiki As Variant: TempHiki = GetRecordsArr(filePath, tableName, SQLfilter)
  
  If IsEmpty(TempHiki) Then
    MsgBox "資材重量なし", vbCritical
    Exit Function
  End If
  
  Dim Hikifields As Variant: Hikifields = GetFieldsArr(filePath, tableName, "")
  If IsEmpty(Hikifields) Then
    MsgBox "資材重量なし", vbCritical
    Exit Function
  End If
  
  '====== フィールドインデックスの取得 ======
  Dim fieldIdx As MaterialFieldIndices
  fieldIdx = GetFieldIndices(Hikifields)
  
  '====== HBデータの計算 ======
  Dim hbData As hbData
  hbData = CalculateHBData(CoilText, coilNo, TempHiki, fieldIdx)
  
  '====== 表記用フォームへの反映 ======
  SetHBDisplayForm hbData
  
  '====== シートへの転記 ======
  WriteHBToSheet coilNo, hbData
  
  DB_HB = hbData.HBValue
End Function

'========== 1. HBデータの計算 ==========
Public Function CalculateHBData(CoilText As Double, coilNo As Long, _
                                  TempHiki As Variant, fieldIdx As MaterialFieldIndices) As hbData
  Dim HB As hbData
  
  '====== コイル寸法の取得 ======
  GetHBCoilDimensions HB, CoilText
  
  '====== HBマスタデータの取得 ======
  GetHBMaterialData HB, TempHiki, fieldIdx
  
  '====== HBフラグの判定 ======Flag が HB.HBFlag に入る
  HB.HBFlag = JudgeHBFlag(CoilText)
  
  '====== HB重量の計算表示 ======HB.HBFlag を参照
  CalculateHBValue HB, CoilText
  
  CalculateHBData = HB
End Function

'========== 共通：コイル高さ TA（m単位）==========
' TA = コイル積高 + チップボール積厚（本数+1枚 × 0.7mm）変数化済
Public Function GetCoilTA(CoilText As Double, col As Double) As Double
  Dim TIPT As Double
  TIPT = val((CoilText + CHIP_EXTRA_SHEETS) * CHIP_THICKNESS_MM / 10 ^ 3)   ' m
  GetCoilTA = val(CoilText * col / 10 ^ 3) + TIPT                           ' m
End Function

Public Function GetCoilGAI() As Double
  Dim PIWork As Double: PIWork = WorksheetFunction.PI()
  With テスラ指定サイズ
    If .TIP1000 Then
      GetCoilGAI = val(1100 * PIWork / 10 ^ 3)
    ElseIf .TIP950 Then
      GetCoilGAI = val(1000 * PIWork / 10 ^ 3)
    ElseIf .TIP820 Then
      GetCoilGAI = val(900 * PIWork / 10 ^ 3)
    End If
  End With
End Function

  HB.HB530固定値 = GetUnitMassAndCoefficient(TempHiki, "HB530*2000", fieldIdx.梱包資材名, fieldIdx.単位質量, fieldIdx.係数)  ' 2026.3.26

'========== 4. HBフラグの判定 ==========
Public Function JudgeHBFlag(CoilText As Double) As Boolean
  Dim Flag As Boolean: Flag = False
  
  With テスラ指定サイズ
    Select Case True
      Case (.CheckBox1) And (CoilText = 14): Flag = True
      Case (.CheckBox2) And (CoilText = 15): Flag = True
      Case (.CheckBox3) And (CoilText = 8): Flag = True
      Case (.CheckBox3) And (CoilText = 7): Flag = True
      Case (.CheckBox4) And (CoilText = 7): Flag = True
      Case (.CheckBox4) And (CoilText = 6): Flag = True
      Case (.CheckBox5) And (CoilText = 11): Flag = True          ' 2026.3.26
      Case (.CheckBox5) And (CoilText = 10): Flag = True          ' 2026.3.26
      Case (.CheckBox6) And (CoilText = 9): Flag = True
      Case (.CheckBox6) And (CoilText = 8): Flag = True
      Case (.ck_08_53) And (CoilText = 11): Flag = True            ' 2026.3.26
      Case (.ck_08_53) And (CoilText = 10): Flag = True            ' 2026.3.26
      ' OB モードも同様に本数条件付き_2026.3.26
      Case (.OptionButton5 Or .OptionButton6) And (CoilText = 11): Flag = True
      Case (.OptionButton5 Or .OptionButton6) And (CoilText = 10): Flag = True
      Case (.lblSpec_08_53_1 Or .lblSpec_08_53_2) And (CoilText = 11): Flag = True
      Case (.lblSpec_08_53_1 Or .lblSpec_08_53_2) And (CoilText = 10): Flag = True
      Case Else: Flag = False
    End Select
  End With
  
  JudgeHBFlag = Flag
End Function

'========== 5. HB重量と計算式の算出 ==========
'Flag=True & col=53.5 & 11本HB590*2000固定値×2 + 八角×2
'Flag=True & col=53.5 & 10本HB530*2000固定値×2 + 八角×2
'Flag=True & それ以外ハードボード(GAI+0.7)×TA×単重 + 八角×2
'Flag=Falseハードボード(GAI+0.7)×TA×単重 + 八角×2

Private Sub CalculateHBValue(ByRef HB As hbData, CoilText As Double)

    If HB.HBFlag And HB.col = 53.5 And CoilText = 11 Then
        '--- 11本: HB590*2000 使用 ---
        HB.HBValue = 2 * HB.HB固定値(0) * HB.HB固定値(1) _
                    + 2 * HB.HB八角用(0) * HB.HB八角用(1)
        HB.HBFormula = "【ハードボード】HB590*2000 上下2枚 + 八角上下2枚" & vbCrLf & _
            "胴巻き: 2 × " & Format(HB.HB固定値(0), "0.000") & _
            " + 八角: 2 × " & Format(HB.HB八角用(0), "0.000") & _
            " = " & Format(HB.HBValue, "0.000")

    ElseIf HB.HBFlag And HB.col = 53.5 And CoilText = 10 Then
        '--- 10本: HB530*2000 使用 ---
        HB.HBValue = 2 * HB.HB530固定値(0) * HB.HB530固定値(1) _
                    + 2 * HB.HB八角用(0) * HB.HB八角用(1)
        HB.HBFormula = "【ハードボード】HB530*2000 上下2枚 + 八角上下2枚" & vbCrLf & _
            "胴巻き: 2 × " & Format(HB.HB530固定値(0), "0.000") & _
            " + 八角: 2 × " & Format(HB.HB八角用(0), "0.000") & _
            " = " & Format(HB.HBValue, "0.000")

    Else
        '--- 上記以外は全て計算HB（Flag の真偽によらず同一式）---
        CalcHBStandard HB
    End If

End Sub

'========== 標準HB（八角上下2枚 + 胴巻き計算値）==========
' 胴巻きHBの板寸法
'   外周方向: 外周 + 700mm  … 巻き重ね代
'   高さ方向: HB_BAND_CLEARANCE_M（現状0）を引く
' 八角HB（上下2枚）はDB固定値のため、この補正の影響を受けない
Private Sub CalcHBStandard(ByRef HB As hbData)
    Dim HBHeight As Double
    HBHeight = GetHBBandHeight(HB.TA)

    HB.HBValue = (HB.GAI + HB_MARGIN_M) * HBHeight * HB.HB計算用(0) * HB.HB計算用(1) _
                + 2 * HB.HB八角用(0) * HB.HB八角用(1)

    HB.HBFormula = "【ハードボード】八角上下2枚+(外周+" & Format(HB_MARGIN_M * 1000, "0") & "mm)×コイル積高×HB単重" & vbCrLf & _
        "2 × " & Format(HB.HB八角用(0), "0.000") & _
        " + (" & Format(HB.GAI, "0.000") & " + " & HB_MARGIN_M & ") × " & _
        Format(HBHeight, "0.000") & " × " & Format(HB.HB計算用(0), "0.000") & _
        " = " & Format(HB.HBValue, "0.000")
End Sub

'========== 胴巻きHBの高さ ==========
' HB_BAND_CLEARANCE_M が 0 の間は積高をそのまま返す
' 補正後が下限を割る場合も積高をそのまま返す（実用上は発生しない想定）
Private Function GetHBBandHeight(TA As Double) As Double
    Dim h As Double
    h = TA - HB_BAND_CLEARANCE_M
    If h < HB_MIN_HEIGHT_M Then
        GetHBBandHeight = TA
    Else
        GetHBBandHeight = h
    End If
End Function

Function GetUnitMassAndCoefficient(TempHiki As Variant, targetName As String, H_梱包資材名 As Long, H_単位質量 As Long, H_係数 As Long) As Variant
'TempHiki(i, H_梱包資材名)がオブジェクト名と一致した場合に、TempHiki(i, H_単位質量)とTempHiki(i, H_係数)を返す関数
Dim i As Long, result(1) As Variant
' TempHiki の要素数を取得（2次元配列を仮定）
On Error Resume Next
Dim RowCount As Long
RowCount = UBound(TempHiki, 1)
On Error GoTo 0
' 配列をループ
For i = 1 To RowCount
    If TempHiki(i, H_梱包資材名) = targetName Then
        ' 一致した場合、単位質量と係数を取得して返す
        result(0) = TempHiki(i, H_単位質量)
        result(1) = TempHiki(i, H_係数)
        GetUnitMassAndCoefficient = result
        Exit Function
    End If
Next i
GetUnitMassAndCoefficient = Array("", "") ' 一致しない場合は空の配列を返す
End Function

'========== ListViewにDB資材を表示 ==========
Public Sub PopulateListViewMaterials(TempHiki As Variant, fieldIdx As MaterialFieldIndices)
  Dim lv As Object
  Set lv = 表記用.ListViewMaterials  '← 実際のコントロール名に変更
  
  '既存の項目をクリア
  lv.ListItems.Clear
  lv.ColumnHeaders.Clear
  
  'リスト詳細ビューを設定
  lv.View = 3
  lv.FullRowSelect = True
  lv.Gridlines = True
  
  '====== 列ヘッダーを追加 ======
  lv.ColumnHeaders.Add , , "梱包資材名", 180
  lv.ColumnHeaders.Add , , "単位質量", 100
  
  '====== DBから行を追加（可変対応） ======
  Dim i As Long, j As Long
  
  'TempHikiの行数をループ
  For i = LBound(TempHiki, 1) To UBound(TempHiki, 1)
    Dim li As Object
    Dim 資材名 As String, 単位質量 As String
    
    '資材名と単位質量を取得
    資材名 = CStr(TempHiki(i, fieldIdx.梱包資材名))
    単位質量 = CStr(TempHiki(i, fieldIdx.単位質量))
    
    'ListItemを追加
    Set li = lv.ListItems.Add()
    li.text = 資材名
    li.SubItems(1) = 単位質量
    
    Debug.Print "行" & i & ": " & 資材名 & " - " & 単位質量
  Next i
  
  'カラム幅を自動調整
  AutoFitListViewColumns lv
  
  Debug.Print "ListViewに" & lv.ListItems.count & "件のデータを追加しました"
End Sub

'========== ListViewのカラム幅を自動調整 ==========
Public Sub AutoFitListViewColumns(lv As Object)
  Dim i As Long
  
  '固定幅を設定
  For i = 1 To lv.ColumnHeaders.count
    If i = 1 Then
      lv.ColumnHeaders(i).width = 150  '梱包資材名
    Else
      lv.ColumnHeaders(i).width = 85  '単位質量
    End If
  Next i
End Sub
